function convertQualityLabel(w)
{
    var ql = '';
    if (w >= 4100 && w < 8000) {
        ql = '4320';
    } else if (w >= 3200 && w < 4100) {
        ql = '2160';
    } else if (w >= 2100 && w < 3200) {
        ql = '1440';
    } else if (w >= 1500 && w < 2100) {
        ql = '1080';
    } else if (w >= 1000 && w < 1500) {
        ql = '720';
    } else if (w >= 750 && w < 1000) {
        ql = '480';
    } else if (w >= 550 && w < 750) {
        ql = '360';
    } else if (w >= 350 && w < 550) {
        ql = '240';
    } else if (w >= 150 && w < 350) {
        ql = '144';
    }
    qlLabel = ql + 'p';
    if (ql == '4320') {
        qlLabel += ' (8K)';
    } else if (ql == '2160') {
        qlLabel += ' (4K)';
    } else if (ql == '1440') {
        qlLabel += ' (2K)';
    }
    return qlLabel;
}
